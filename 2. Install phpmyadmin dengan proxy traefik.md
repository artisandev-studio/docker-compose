## Install PHPMyAdmin, PHP8.3-fpm, dan MySQL 8.0
### 1. Buat Folder
```bash
mkdir phpmyadmin
```

### 2. Buat file .env
```bash
sudo nano .env
```
Diisi dengan :
```bash
PMA_DOMAIN=sub.domain.com
MYSQL_ROOT_PASSWORD=PasswordRoot
```
### 3. Buat docker-compose.yml
```
sudo nano .env
```
Diisi dengan :
```bash
services:
  # 1. Database MySQL 8.0
  db:
    image: mysql:8.0
    container_name: mysql_db
    restart: always
    environment:
      MYSQL_ROOT_PASSWORD: "${MYSQL_ROOT_PASSWORD}"
    volumes:
      - db_data:/var/lib/mysql
    networks:
      - db_internal

  # 2. Prosesor PHP (FPM)
  pma_fpm:
    image: phpmyadmin:fpm-alpine
    container_name: pma_fpm
    restart: always
    environment:
      - PMA_HOST=mysql_db
    volumes:
      # Source code phpMyAdmin ditaruh di volume agar bisa dibaca Nginx
      - pma_code:/var/www/html
    networks:
      - db_internal   # Jaringan ke DB
      - pma_internal  # Jaringan ke Nginx

  # 3. Web Server Nginx (Berhadapan dengan Traefik)
  pma_web:
    image: nginx:alpine
    container_name: pma_nginx
    restart: always
    depends_on:
      - pma_fpm  # PENTING: Nginx akan menunggu FPM menyala dan mengisi volume pma_code
    volumes:
      - pma_code:/var/www/html:ro
      - ./nginx.conf:/etc/nginx/conf.d/default.conf:ro
    networks:
      - pma_internal  # Jaringan ke FPM
      - webproxy      # Jaringan ke Traefik
    labels:
      - "traefik.enable=true"
      - "traefik.http.routers.pma.rule=Host(`${PMA_DOMAIN}`)"
      - "traefik.http.routers.pma.entrypoints=websecure"
      - "traefik.http.routers.pma.tls.certresolver=letsencrypt"
      - "traefik.http.services.pma.loadbalancer.server.port=80"
      - "traefik.docker.network=webproxy"

volumes:
  db_data:
  pma_code:

networks:
  db_internal:
  pma_internal:
  webproxy:
    external: true
```

### 4. Buat file confi nginx
```bash
sudo nano nginx.conf
```
Disi dengan :
```bash
server {
    listen 80;
    server_name _;
    root /var/www/html;
    index index.php index.html;

    location / {
        try_files $uri $uri/ /index.php?$args;
    }

    location ~ \.php$ {
        fastcgi_pass pma_fpm:9000; # Harus sama dengan nama container FPM di bawah
        fastcgi_index index.php;
        include fastcgi_params;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
    }
}
```
### 5. Jalankan docker compose
```bash
sudo docker compose up -d
```
